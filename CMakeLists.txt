cmake_minimum_required(VERSION 3.24)
project(ObsidianLauncher VERSION 2.3.0 LANGUAGES C CXX)
include(GNUInstallDirs)
set(CMAKE_INSTALL_BINDIR "." CACHE PATH "Portable app directory" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Widgets Concurrent Network)
include(FetchContent)
# Fixed upstream release; only downloaded when configuring a fresh build.
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON CACHE BOOL "Disable zlib's standalone install rules" FORCE)
FetchContent_Declare(zlib URL https://zlib.net/fossils/zlib-1.3.1.tar.gz
    URL_HASH SHA256=9a93b2b7dfdac77ceba5a558a580e74667dd6fede4585b91eefb60f03b72df23)
FetchContent_MakeAvailable(zlib)
add_library(launcher_core STATIC src/Minecraft.cpp src/Minecraft.h src/NativeArchive.cpp src/NativeArchive.h src/LogRedactor.h)
target_link_libraries(launcher_core PUBLIC Qt6::Core PRIVATE zlibstatic)
target_include_directories(launcher_core PUBLIC src PRIVATE ${zlib_SOURCE_DIR} ${zlib_BINARY_DIR})
add_library(launcher_ui STATIC src/DiscordPresence.cpp src/DiscordPresence.h src/AuthClient.cpp src/AuthClient.h src/PathLabel.h src/MainWindow.cpp src/MainWindow.h src/LogWindow.cpp src/LogWindow.h src/Theme.h src/Appearance.h src/Preferences.cpp src/Preferences.h src/Ui.cpp src/Ui.h src/WorldBanner.h src/resources.qrc)
target_link_libraries(launcher_ui PUBLIC launcher_core Qt6::Widgets Qt6::Concurrent Qt6::Network)
add_executable(ObsidianLauncher WIN32 src/main.cpp)
if(WIN32)
    target_sources(ObsidianLauncher PRIVATE src/app.rc)
endif()
target_link_libraries(ObsidianLauncher PRIVATE launcher_ui)
if(MSVC)
    target_compile_options(launcher_core PRIVATE /W4 /utf-8)
    target_compile_options(ObsidianLauncher PRIVATE /W4 /utf-8)
    target_compile_options(launcher_ui PRIVATE /W4 /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(LauncherTests tests/LauncherTests.cpp)
    target_link_libraries(LauncherTests PRIVATE launcher_core Qt6::Test zlibstatic)
    target_include_directories(LauncherTests PRIVATE ${zlib_SOURCE_DIR} ${zlib_BINARY_DIR})
    add_test(NAME LauncherTests COMMAND LauncherTests)
    add_executable(MockJava tests/MockJava.cpp)
    target_link_libraries(MockJava PRIVATE Qt6::Core)
    add_executable(UiTests tests/UiTests.cpp)
    target_link_libraries(UiTests PRIVATE launcher_ui Qt6::Test)
    if(MSVC)
        target_compile_options(LauncherTests PRIVATE /utf-8)
        target_compile_options(UiTests PRIVATE /utf-8)
    endif()
    add_executable(AuthTests tests/AuthTests.cpp)
    target_link_libraries(AuthTests PRIVATE launcher_ui Qt6::Test)
    if(MSVC)
        target_compile_options(AuthTests PRIVATE /utf-8)
    endif()
    add_executable(DiscordTests tests/DiscordTests.cpp)
    target_link_libraries(DiscordTests PRIVATE launcher_ui Qt6::Test)
    if(MSVC)
        target_compile_options(DiscordTests PRIVATE /utf-8)
    endif()
    add_test(NAME DiscordTests COMMAND DiscordTests)
    add_test(NAME AuthTests COMMAND AuthTests -platform windows)
    set_tests_properties(AuthTests PROPERTIES ENVIRONMENT "MOCK_JAVA_EXE=$<TARGET_FILE:MockJava>" TIMEOUT 60)
    add_test(NAME UiTests COMMAND UiTests -platform windows)
    set_tests_properties(UiTests PROPERTIES ENVIRONMENT "MOCK_JAVA_EXE=$<TARGET_FILE:MockJava>" TIMEOUT 45)
endif()
install(TARGETS ObsidianLauncher RUNTIME DESTINATION .)
install(FILES ${zlib_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME zlib.txt)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY licenses/ DESTINATION licenses)
install(DIRECTORY docs/ DESTINATION docs)
qt_generate_deploy_app_script(TARGET ObsidianLauncher OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
